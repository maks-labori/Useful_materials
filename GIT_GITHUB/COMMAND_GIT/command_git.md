# Команды Git/Github

```bash
# если git на устройстве настраивается впервые, появится предупреждение и потребуются команды:
git config --global user.name "Имя Ф."
git config --global user.email "email@mail.ru"
```

## команды git для старта проекта

```bash
git init #(инициализация гит)

git branch -m main #переименовываем ветку(по умолчанию master)

git remote add origin <ссылка на репозиторий вида URL.git>

git add <название файла или . - все> #пример git add project.cpp

git commit -m "<название коммита>"

git push origin main #отправить локальные изменения в репозиторий
```

## остальные полезные команды
```bash
git status #посмотреть состояния файлов(готовы к коммиту,удаленные и тд)

git log #посмотреть историю комитов
git log --oneline --graph #уменьшения длины вывода комита и вывод в виде графа веток

git pull origin main #если нужно наооборот подтянуть изменения оттуда

git reset <имя файла> #отменить git add

git clone <URL>.git #взять чужой репозиторий себе полностью

git branch <имя ветки> #создать новую ветку

git branch #посмотреть список веток

git branch -d <имя_ветки>  #удалить ветку после слияния
git branch -D <имя_ветки>  #просто удалить со всем содержимым

git checkout <имя ветки> #перейти в ветку
git switch <имя ветки> #аналог

git checkout <хэш номер коммита> #перейти к комиту

git checkout -b <имя ветки> #создать ветку и сразу перейти в нее

git stash #убрать нанынешние изменения в "карман"
git stash pop #достать из "кармана"
git stash list #посмотреть список "заначек"

git merge <назввание ветки> #сливает ветки в одну(вливаем именно в текущую)
```

## полезные терминал команды

```bash
cd <путь к папке> #перейти к папке,можно общий путь,можно относительный
cd .. #назад по директории
cd ~ #назад к самому началу

mkdir <имя папки> #создать папку

rmdir <имя папки> #удалить папку
rm -rf <имя папки> #удалить вместе с файлами
rm <имя файла> #удалить файл

code <имя файла> #запустить файл в ide

dir #посмотреть все файлы внутри этой папки
ls #аналог

ls -a #посмотреть даже скрытые файлы

cat <имя файла> #открыть файл в терминале
```

## разрешение конфликтов

```bash
git rm -r --cached . #если поменял гитигнор,но папка осталась в кеше

#удаление коммитов
git reset --soft HEAD~1 #сохраняет файлы
git reset --hard HEAD~1 #не сохраняет

#удаление с гитхаба
git revert HEAD #делает еще 1 комит,отменяющий предыдущий
git push --force #насильно перезаписывает комиты с гитхаба на локальные
```
